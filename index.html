# Find a pattern, not a new framework

# Read the exact styling vocabulary
$ node bin/sprig.js contract

# Catch a few common HTML mistakes
$ node bin/sprig.js check page.html

# Pack only the modules your HTML uses
$ node bin/sprig.js build --html page.html
